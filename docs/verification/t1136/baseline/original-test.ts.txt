async function dispatch<T = {id:string}>(page:Page,id:string,input:unknown,scope:Scope={}):Promise<T> {
 const response=await page.request.post("/api/v1/dispatch",{...wait,headers:{"content-type":"application/json",...(scope.groupId?{"x-group-id":scope.groupId}:{})},data:{id,input}});
 expect(response.ok(),id+" HTTP "+response.status()).toBe(true);
 return (await response.json() as {data:T}).data;
}
async function createGroup(page:Page,name:string) {
 return dispatch(page,"accountGroups.create",{body:{name,idempotencyKey:crypto.randomUUID()}});
}
async function transfer(page:Page,accountId:string,toGroupId:string) {
 const preview=await dispatch<Record<string,unknown>>(page,"accountGroups.members.previewTransfer",{body:{accountId,toGroupId}});
 await dispatch(page,"accountGroups.members.transfer",{body:{...preview,idempotencyKey:crypto.randomUUID()}});
}

function today() {
 const parts=new Intl.DateTimeFormat("en-CA",{timeZone:"Asia/Shanghai",year:"numeric",month:"2-digit",day:"2-digit"}).formatToParts(new Date());
 return ["year","month","day"].map(type=>parts.find(p=>p.type===type)!.value).join("-");
}
async function list(page:Page,groupId:string) {
 const day=today(),month=day.slice(0,7),year=Number(month.slice(0,4)),m=Number(month.slice(5));
 const end=new Date(Date.UTC(year,m,0)).toISOString().slice(0,10);
 return dispatch<Entry[]>(page,"group.calendar.list",{params:{groupId},query:{from:month+"-01T00:00:00.000Z",to:end+"T23:59:59.000Z",limit:100,offset:0}},{groupId});
}
async function date(page:Page,dialog:Locator) {

 const cell=page.locator('.ant-picker-dropdown:not(.ant-picker-dropdown-hidden) td[title="'+today()+'"]');
 await expect(cell).toBeVisible(wait);await cell.click(wait);
 await expect(dialog.getByLabel("创作日期")).toHaveValue(today(),wait);
}
async function openDay(page:Page) {

 await expect(page.getByTestId("calendar-day-drawer")).toBeVisible(wait);
}
async function dimensions(dialog:Locator) {
 return dialog.evaluate(node=>{
  const rect=node.getBoundingClientRect(),style=getComputedStyle(node);
  const controls=Array.from(node.querySelectorAll("select,button,input")).filter(el=>getComputedStyle(el).position!=="absolute").map(el=>{
   const box=el.getBoundingClientRect();
   return {label:el.getAttribute("aria-label")||el.textContent?.trim(),left:box.left,right:box.right,width:box.width};
  });
  return {viewport:innerWidth,left:rect.left,right:rect.right,width:rect.width,scrollWidth:node.scrollWidth,clientWidth:node.clientWidth,maxWidth:style.maxWidth,gridTemplateColumns:style.gridTemplateColumns,controls};
 });
}
function noOverflow(d:Awaited<ReturnType<typeof dimensions>>) {
 return d.left>=15&&d.right<=d.viewport-15&&d.scrollWidth<=d.clientWidth+1&&d.controls.every(c=>c.left>=d.left-1&&c.right<=d.right+1);
}
async function capture(page:Page,dialog:Locator,name:string) {
 mkdirSync(resolve(evidence,"screenshots"),{recursive:true});
 await page.screenshot({path:resolve(evidence,"screenshots",name+".png")});
 return {name,url:page.url(),viewport:page.viewportSize(),dimensions:await dimensions(dialog),text:await dialog.innerText()};
}
